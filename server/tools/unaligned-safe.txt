# Casts checked by hand (tools/check-unaligned.sh -s): what keeps each aligned.
# libc.c memcpy: only taken when both pointers are 4-aligned (tested first).
mod/libc.c:.*\(\(uint32_t \*\) dp\)\[[0-3]\] = \(\(const uint32_t \*\) sp\)
mod/libc.c:.*\*\(uint32_t \*\) dp = \*\(const uint32_t \*\) sp;
# libc.c memset: the loop before it brings dp to a 4-byte boundary.
mod/libc.c:.*\*\(uint32_t \*\) dp = w;
# libc.c block_size: our own 8-byte header in front of every heap block.
mod/libc.c:.*\(const size_t \*\) \(\(const char \*\) p - BLOCK_HEAD\)
# modclient.c: the module's table is at &34 from a module's start, and
# modules are word-aligned in the RMA.
modclient.c:.*\(const rdps_mod_table \*\) \(base \+ RDPS_MOD_TABLE\)
